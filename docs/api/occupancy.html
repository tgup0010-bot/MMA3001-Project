 44    """Resample an occupancy event log onto a regular per-room time grid.
 45
 46    Args:
 47        df: Occupancy event log as returned by
 48            :func:`occupancy.data_loading.load_occupancy_log`. Must contain
 49            ``room_col``, ``time_col`` and ``status_col``.
 50        bin_size: Pandas frequency string for the output grid, e.g.
 51            ``"15min"``, ``"30min"``, ``"1h"``.
 52        occupied_statuses: Which values of ``status_col`` count as
 53            "occupied" for the binary label. Defaults to only
 54            ``"CurrentlyOccupied"`` (a strict definition); pass
 55            ``("CurrentlyOccupied", "RecentlyOccupied")`` for a looser one
 56            that also counts the sensor's post-departure cooldown state as
 57            occupied.
 58        max_gap: The longest real gap between consecutive observations
 59            that is still forward-filled. Grid points falling in a larger
 60            gap are set to ``NaN`` (missing) rather than assumed occupied
 61            or unoccupied.
 62        room_col: Column identifying the room/zone.
 63        time_col: Column holding the (timezone-aware) event timestamp.
 64        status_col: Column holding the raw occupancy status string.
 65
 66    Returns:
 67        A tidy DataFrame with one row per (room, time bin), columns
 68        ``[room_col, time_col, "occupied"]``, where ``"occupied"`` is
 69        ``1.0``, ``0.0``, or ``NaN`` (gap longer than ``max_gap``).
 70
 71    Notes:
 72        Rooms are processed independently (looped in Python) rather than
 73        vectorised across the whole file at once. With only 5 rooms in
 74        this dataset the loop overhead is negligible, and it keeps the
 75        per-room grid construction (which depends on that room's own
 76        min/max timestamp) simple and easy to verify.
 77    """
 78    max_gap = pd.Timedelta(max_gap)
 79
 80    work = (
 81        df[[room_col, time_col, status_col]]
 82        .dropna(subset=[time_col])
 83        .copy()
 84    )
 85    work["occupied"] = work[status_col].isin(occupied_statuses).astype(float)
 86
 87    # If a room logged two events at the exact same timestamp, keep the
 88    # later-recorded one as the authoritative state at that instant.
 89    work = work.sort_values([room_col, time_col]).drop_duplicates(
 90        subset=[room_col, time_col], keep="last"
 91    )
 92
 93    grids = []
 94    for room, g in work.groupby(room_col, sort=False):
 95        g = g.sort_values(time_col)
 96        tz = g[time_col].dt.tz
 97        start = g[time_col].min().floor(bin_size)
 98        end = g[time_col].max().ceil(bin_size)
 99        grid = pd.DataFrame(
100            {time_col: pd.date_range(start, end, freq=bin_size, tz=tz)}
101        )
102
103        # merge_asof with a tolerance is the vectorised equivalent of
104        # "look up the most recent known state, but only if it's not too
105        # stale" -- this is what enforces max_gap.
106        merged = pd.merge_asof(
107            grid,
108            g[[time_col, "occupied"]],
109            on=time_col,
110            direction="backward",
111            tolerance=max_gap,
112        )
113        merged[room_col] = room
114        grids.append(merged)
115
116    out = pd.concat(grids, ignore_index=True)
117    return out[[room_col, time_col, "occupied"]]

 61    resampled: pd.DataFrame,
 62    *,
 63    room_col: str = "floorspaceid",
 64    time_col: str = "collecteddate",
 65    status_col: str = "occupied",
 66    rolling_window: int = 4,
 67) -> tuple[pd.DataFrame, dict]:
 68    """Build a supervised (features + target) table from the resampled grid.
 69
 70    Args:
 71        resampled: Output of
 72            :func:`occupancy.preprocessing.resample_occupancy`.
 73        room_col: Column identifying the room/zone.
 74        time_col: Column holding the bin's start time.
 75        status_col: Binary/`NaN` occupancy column (as produced by
 76            ``resample_occupancy``).
 77        rolling_window: Number of trailing bins (including the current
 78            one) averaged into ``rolling_mean_1h`` -- 4 bins at the default
 79            15-minute bin size covers the last hour.
 80
 81    Returns:
 82        A tuple ``(table, info)`` where:
 83
 84        - ``table`` has columns ``room_col``, ``time_col``, every name in
 85          :data:`NUMERIC_FEATURES`, and ``target`` (the *next* bin's
 86          occupancy, 0.0/1.0) -- rows with any missing feature or target
 87          have been dropped (see ``info``).
 88        - ``info`` reports how many rows were dropped and why, so this is
 89          visible in the report rather than silently absorbed. Rows are
 90          dropped only because the underlying sensor data was itself
 91          missing (see :func:`occupancy.preprocessing.resample_occupancy`'s
 92          ``max_gap``) -- this function does not impute or guess a
 93          missing occupancy state, since doing so would make the model
 94          circular (predicting an occupancy value manufactured by the
 95          feature pipeline itself).
 96
 97    Raises:
 98        ValueError: If ``resampled`` is empty after grouping, e.g. because
 99            ``room_col`` has no rows.
100    """
101    if resampled.empty:
102        raise ValueError("resample_occupancy output is empty; nothing to build features from.")
103
104    df = resampled.sort_values([room_col, time_col]).reset_index(drop=True).copy()
105    df = _add_calendar_features(df, time_col)
106
107    grp = df.groupby(room_col)[status_col]
108    df["occupied_now"] = df[status_col]
109    df["occupied_lag1"] = grp.shift(1)
110    df["rolling_mean_1h"] = grp.transform(
111        lambda s: s.rolling(rolling_window, min_periods=rolling_window).mean()
112    )
113    # The target is the *next* bin's state -- shift(-1) looks forward, which
114    # is exactly what we want to predict, and is only safe because it's
115    # computed per-room (groupby) so a room boundary never leaks into
116    # another room's "next" value.
117    df["target"] = grp.shift(-1)
118
119    feature_cols = list(NUMERIC_FEATURES)
120    needed = feature_cols + ["target"]
121
122    n_before = len(df)
123    table = df.dropna(subset=needed).reset_index(drop=True)
124    n_after = len(table)
125
126    info = {
127        "rows_before": n_before,
128        "rows_after": n_after,
129        "rows_dropped": n_before - n_after,
130        "pct_dropped": 100.0 * (n_before - n_after) / n_before if n_before else 0.0,
131    }
132
133    return table[[room_col, time_col, *feature_cols, "target"]], info

class MarkovBaseline: 
 32class MarkovBaseline:
 33    """First-order Markov chain, conditioned on room and hour-of-day.
 34
 35    Estimates ``P(occupied_next=1 | room, hour_of_day, occupied_now)``
 36    directly from training data by counting/averaging observed
 37    transitions, then looks up that probability at prediction time. This
 38    is a classical statistical/numerical baseline: no gradient-based
 39    fitting, no hyperparameters beyond the discretisation already fixed by
 40    the time-grid bin size.
 41    """
 42
 43    def __init__(self) -> None:
 44        self.table_: pd.Series | None = None
 45        self.global_rate_: float | None = None
 46
 47    def fit(
 48        self,
 49        table: pd.DataFrame,
 50        *,
 51        room_col: str = "floorspaceid",
 52        time_col: str = "collecteddate",
 53    ) -> "MarkovBaseline":
 54        """Estimate transition probabilities from a training table.
 55
 56        Args:
 57            table: Training supervised table (see
 58                :func:`occupancy.features.build_supervised_dataset`); must
 59                contain ``room_col``, ``time_col``, ``occupied_now`` and
 60                ``target``.
 61            room_col: Column identifying the room/zone.
 62            time_col: Column holding the bin's start time.
 63
 64        Returns:
 65            ``self``, fitted.
 66        """
 67        d = table.copy()
 68        d["hour"] = d[time_col].dt.hour
 69        self.global_rate_ = float(d["target"].mean())
 70        self.table_ = (
 71            d.groupby([room_col, "hour", "occupied_now"])["target"]
 72            .mean()
 73            .rename("p_next_occupied")
 74        )
 75        return self
 76
 77    def predict_proba(
 78        self,
 79        table: pd.DataFrame,
 80        *,
 81        room_col: str = "floorspaceid",
 82        time_col: str = "collecteddate",
 83    ) -> np.ndarray:
 84        """Predict P(occupied at next bin) for each row of ``table``.
 85
 86        A (room, hour, occupied_now) combination never seen during
 87        ``fit`` falls back to the overall training-set occupancy rate
 88        (``global_rate_``) rather than raising or returning ``NaN`` --
 89        this can genuinely happen for a room/hour combination with very
 90        little training data, and failing silently to a sensible default
 91        is preferable to crashing evaluation.
 92
 93        Args:
 94            table: Table to predict on; same column requirements as
 95                ``fit``.
 96            room_col: Column identifying the room/zone.
 97            time_col: Column holding the bin's start time.
 98
 99        Returns:
100            Predicted probabilities, one per row of ``table``.
101
102        Raises:
103            RuntimeError: If called before :meth:`fit`.
104        """
105        if self.table_ is None:
106            raise RuntimeError("MarkovBaseline.predict_proba called before fit().")
107
108        d = table.copy()
109        d["hour"] = d[time_col].dt.hour
110        keys = pd.MultiIndex.from_frame(d[[room_col, "hour", "occupied_now"]])
111        p = self.table_.reindex(keys).to_numpy()
112        return np.where(np.isnan(p), self.global_rate_, p)

def build_logistic_pipeline(**logistic_regression_kwargs) -> sklearn.pipeline.Pipeline: 
23def build_logistic_pipeline(**logistic_regression_kwargs) -> Pipeline:
24    """Construct the preprocessing + classifier pipeline.
25
26    Args:
27        **logistic_regression_kwargs: Passed through to
28            `sklearn.linear_model.LogisticRegression` (e.g. ``C=0.5``),
29            letting callers tune regularisation strength during the
30            sensitivity-analysis step without editing this function.
31
32    Returns:
33        An unfitted `sklearn.pipeline.Pipeline` with:
34
35        1. one-hot encoding of ``occupancy.features.CATEGORICAL_FEATURES``
36           (currently just the room identifier), with unknown rooms at
37           prediction time encoded as all-zero rather than raising; and
38        2. a `sklearn.linear_model.LogisticRegression` over that plus the
39           passthrough numeric features
40           (``occupancy.features.NUMERIC_FEATURES``).
41    """
42    preprocessor = ColumnTransformer(
43        transformers=[
44            (
45                "room",
46                OneHotEncoder(handle_unknown="ignore"),
47                list(CATEGORICAL_FEATURES),
48            ),
49        ],
50        remainder="passthrough",  # numeric features pass through unscaled;
51        # logistic regression's L2 penalty is somewhat scale-sensitive, but
52        # all numeric features here are already on comparable [0,1]/[-1,1]
53        # scales (occupancy fractions, sin/cos), so an explicit scaler was
54        # judged unnecessary complexity for this feature set.
55    )
56    classifier = LogisticRegression(max_iter=1000, **logistic_regression_kwargs)
57    return Pipeline([("preprocess", preprocessor), ("classify", classifier)])

60def build_random_forest_pipeline(**random_forest_kwargs) -> Pipeline:
61    """Construct a random-forest alternative to the logistic regression model.
62
63    Included as a second ML model (rather than treating logistic
64    regression as the only option) so the "alternative solutions"
65    comparison covers two genuinely different modelling families, not just
66    a classical baseline vs. one ML model. A random forest can capture
67    non-linear interactions between features (e.g. "weekday AND
68    afternoon" mattering differently than either alone) that a linear
69    logistic model cannot represent directly.
70
71    Args:
72        **random_forest_kwargs: Passed through to
73            `sklearn.ensemble.RandomForestClassifier` (e.g.

def chronological_split( table: pandas.DataFrame, *, time_col: str = 'collecteddate', train_frac: float = 0.8) -> tuple[pandas.DataFrame, pandas.DataFrame, pandas.Timestamp]: 
16def chronological_split(
17    table: pd.DataFrame,
18    *,
19    time_col: str = "collecteddate",
20    train_frac: float = 0.8,
21) -> tuple[pd.DataFrame, pd.DataFrame, pd.Timestamp]:
22    """Split a table into train/test sets by time, not at random.
23
24    All rows up to a single global cutoff timestamp (computed from
25    ``train_frac`` of the pooled, sorted timestamps across every room) go
26    to training; everything after goes to test. Using one global cutoff
27    (rather than splitting each room independently) keeps the train/test
28    boundary at the same wall-clock moment for every room, which is what
29    "can this model generalise to the future" should mean here.
30
31    Args:
32        table: Supervised table (e.g. from
33            :func:`occupancy.features.build_supervised_dataset`).
34        time_col: Column holding each row's timestamp.
35        train_frac: Fraction of rows (by time, not room-balanced count)
36            assigned to training.
37
38    Returns:
39        ``(train, test, cutoff_time)``.
40
41    Raises:
42        ValueError: If ``train_frac`` is not strictly between 0 and 1, or
43            the split would leave either set empty.
44    """
45    if not 0.0 < train_frac < 1.0:
46        raise ValueError(f"train_frac must be in (0, 1), got {train_frac}")
47
48    sorted_table = table.sort_values(time_col).reset_index(drop=True)
49    cutoff_idx = int(len(sorted_table) * train_frac)
50    if cutoff_idx == 0 or cutoff_idx >= len(sorted_table):
51        raise ValueError(
52            f"train_frac={train_frac} produced an empty train or test set "
53            f"for {len(sorted_table)} rows."
54        )
55
56    cutoff_time = sorted_table.loc[cutoff_idx, time_col]
57    train = sorted_table.iloc[:cutoff_idx].reset_index(drop=True)
58    test = sorted_table.iloc[cutoff_idx:].reset_index(drop=True)
59    return train, test, cutoff_time
